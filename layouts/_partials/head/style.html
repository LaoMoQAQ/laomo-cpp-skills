{{/* 覆盖主题的 head/style.html 【为什么需要覆盖】 --------------------------------------------------------------------------- 主题原本是： {{ $style := $sass | toCSS | minify | resources.Fingerprint "sha256" }} toCSS 走 libsass，只有 Hugo extended 版才有。 Cloudflare 构建镜像装的是标准版 Hugo（版本号无 +extended），构建会失败： TOCSS: failed to transform "/scss/style.scss" (text/x-scss). Check your Hugo installation; you need the extended version to build SCSS/SASS with transpiler set to 'libsass' 【策略：由环境变量决定转译器】 HUGO_SASS_TRANSPILER=dartsass → 用外部 Dart Sass（标准版 Hugo 可用） HUGO_SASS_TRANSPILER=libsass → 用内置 libsass（extended 版可用） 未设置 → 默认 libsass（保持主题原行为） 为什么默认 libsass： 这样本地用 extended 版 Hugo 直接 hugo server 就能跑，不需要额外装东西。 Cloudflare 上由 bin/build.sh 设置 HUGO_SASS_TRANSPILER=dartsass， 并负责安装 Dart Sass，从而绕过标准版 Hugo 没有 libsass 的限制。 outputStyle 用 compressed 替代原来的 minify 管道，产物更小且无需二次处理。 */}} {{- $sass := resources.Get "scss/style.scss" -}} {{- $transpiler := default "libsass" (getenv "HUGO_SASS_TRANSPILER") -}} {{- $opts := dict "transpiler" $transpiler "outputStyle" "compressed" "targetPath" "scss/style.css" -}} {{- $style := $sass | css.Sass $opts | resources.Fingerprint "sha256" -}}

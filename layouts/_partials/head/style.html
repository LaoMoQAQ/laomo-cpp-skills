{{/* 覆盖主题的 head/style.html =========================================================================== 【问题背景】 主题原本是： {{ $style := $sass | toCSS | minify | resources.Fingerprint "sha256" }} toCSS 走 libsass，只有 Hugo extended 版才内置。 Cloudflare 构建镜像装的是「标准版」Hugo（版本号无 +extended）， 因此构建在编译 assets/scss/style.scss 时失败： TOCSS: failed to transform "/scss/style.scss" (text/x-scss). Check your Hugo installation; you need the extended version to build SCSS/SASS with transpiler set to 'libsass' 以前没暴露是因为仓库里误提交了 resources/_gen/ 编译缓存， Hugo 命中缓存就跳过编译。缓存清理后问题立刻出现。 【这里的做法：三级降级，保证任何环境都能构建】 1. libsass —— extended 版 Hugo 内置，最快 2. dartsass —— 需要构建机上有 sass 命令（bin/build.sh 会装） 3. 预编译 CSS —— 前面都不可用时，直接读 static/scss/prebuilt.css 第 3 级是关键：它让构建在"没有 extended、也没有 Dart Sass"的环境下 依然能成功，只是样式用的是仓库里预编译好的那份。 注意 Hugo 的资源转换是惰性的：css.Sass 本身不报错， 真正的错误在访问 .RelPermalink 时才抛出，所以必须用 try 包住 RelPermalink。 改样式后如果第 3 级被用到，记得重新生成 static/scss/prebuilt.css： bash bin/update-prebuilt.sh */}} {{- $sass := resources.Get "scss/style.scss" -}} {{- $style := false -}} {{- /* 1. libsass（extended 版 Hugo） */ -}} {{- if not $style -}} {{- $cand := css.Sass (dict "transpiler" "libsass" "outputStyle" "compressed") $sass -}} {{- with try $cand.RelPermalink -}} {{- if not .Err -}}{{- $style = $cand -}}{{- end -}} {{- end -}} {{- end -}} {{- /* 2. dartsass（标准版 Hugo + 外部 Dart Sass） */ -}} {{- if not $style -}} {{- $cand := css.Sass (dict "transpiler" "dartsass" "outputStyle" "compressed") $sass -}} {{- with try $cand.RelPermalink -}} {{- if not .Err -}}{{- $style = $cand -}}{{- end -}} {{- end -}} {{- end -}} {{- if $style -}} {{- $style = $style | resources.Fingerprint "sha256" -}} {{- else -}} {{- /* 3. 预编译 CSS 兜底 */ -}} {{- $prebuilt := resources.Get "scss/prebuilt.css" -}} {{- if $prebuilt -}} {{- warnf "SCSS 无法编译（缺 extended Hugo 或 Dart Sass），回退到 static/scss/prebuilt.css。改样式后请重新生成该文件。" -}} {{- $prebuilt = $prebuilt | resources.Fingerprint "sha256" -}} {{- else -}} {{- errorf "SCSS 无法编译，且找不到预编译兜底文件 assets/scss/prebuilt.css" -}} {{- end -}} {{- end -}}
